cmake_minimum_required(VERSION 3.20.0)

# RadiANT heart-rate strap: a node, not a dongle. See src/main.c for why it
# exists and Kconfig for what it is allowed to do.
#
# UNLIKE sim/, THIS NEEDS NO sdk-ant AT ALL. sim/ is built entirely out of
# sdk-ant's profile libraries and its radio, and says so in a FATAL_ERROR;
# this one is built out of radiant_core and radiant_core/src/profiles/, both
# of which are in this repository. That is the whole point of it: it is the
# first image where
# the clean-room stack transmits as a master with a real ANT+ profile on top,
# and a dependency on the proven stack would defeat the demonstration.
#
# radiant_core is appended unconditionally and its backend chosen with
# -DRADIANT_BACKEND, exactly as the root CMakeLists.txt does - both now
# through the shared cmake/radiant_backend.cmake, included below. The
# duplication of *calling* it (rather than sharing one CMakeLists.txt) is
# deliberate and matches sim/'s note: `west build strap` must not depend on
# the dongle's build files existing or being current.
#
# THE DEFAULT IS nrf AND NOT null, and the asymmetry with radiant_core's own
# CMake default is the same one scripts/build_all.ps1 documents: null is right
# for a compile check and wrong for a script whose job is producing images
# somebody flashes. A null-radio strap boots, logs "transmitting", and puts
# nothing on the air - indistinguishable from a dead antenna, and it has cost
# this project a Feather flash and a whole session once already.
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/radiant_backend.cmake")
radiant_select_backend("${CMAKE_CURRENT_SOURCE_DIR}/../radiant_core" "nrf" RADIANT_BACKEND_CONF)

set(strap_backend_conf "${CMAKE_CURRENT_BINARY_DIR}/strap_backend.conf")
file(WRITE "${strap_backend_conf}"
"# Generated by strap/CMakeLists.txt from -DRADIANT_BACKEND. Do not edit.
CONFIG_RADIANT_CORE=y
CONFIG_RADIANT_CORE_ANTR_API=y
${RADIANT_BACKEND_CONF}
")
list(APPEND EXTRA_CONF_FILE "${strap_backend_conf}")

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(radiant_strap)

# The backend actually compiled is the backend that was asked for.
#
# radiant_assert_backend(), shared with the top-level CMakeLists.txt - see
# cmake/radiant_backend.cmake for the full rationale. What it guards against
# is a Kconfig `depends on` going silently unmet, falling back to the null
# backend: a null-radio strap boots, logs "transmitting", and puts nothing on
# the air. It has happened twice on the Nordic side, and is more likely on TI,
# where the cc26xx arm has to depend on SOC_SERIES_CC13X2_CC26X2 because no
# SOC_COMPATIBLE_* symbol exists for the part.
radiant_assert_backend("strap")

target_sources(app PRIVATE src/main.c)

# ── The antr_* adapter ──────────────────────────────────────────────────────
#
# ../src/ant_radio_radiant.c (radiant_api.c, before it moved out of
# radiant_core) is what actually implements antr_init() and the rest of
# src/ant_radio.h on top of radiant_core's modules - CONFIG_RADIANT_CORE=y
# alone no longer pulls it in, since it is application-side now (see
# src/Kconfig.antr_api). strap_backend_conf above forces
# CONFIG_RADIANT_CORE_ANTR_API=y unconditionally, so this is never
# conditional in practice, but target_sources_ifdef() is used anyway to match
# the root CMakeLists.txt exactly and to fail at configure rather than at
# link if that ever changes.
target_sources_ifdef(CONFIG_RADIANT_CORE_ANTR_API app PRIVATE
  ../src/ant_radio_radiant.c)
target_sources_ifdef(CONFIG_RADIANT_SEC_HOST_MESSAGES app PRIVATE
  ../src/ant_sec_host.c)

# ── radiant_core/src/profiles ───────────────────────────────────────────────
#
# THE FIRST NON-TEST CONSUMER OF THESE FILES. Until this application they were
# compiled only by radiant_core/tests/CMakeLists.txt, so every profile in the
# project had only ever run inside a unit test. Nothing in them needed changing
# to run for real, which is the claim this build makes and the reason it is
# worth making.
#
# WHAT THE LINKER SAYS IS NEEDED, WHICH IS MORE THAN THE PAGE NEEDS.
#
# The first four are the heart-rate page itself. The other two are pulled in by
# the seam rather than by the profile: profile_sched.c's slot rotation calls
# into profile_schedule.c to encode the descriptor, and profile_hr_next() calls
# profile_compat_next() unconditionally - the compat client is a *seam*, and
# whether it emits anything is a run-time question of whether a compat context
# was attached, not a compile-time one. This application attaches none, so
# profile_hr_set_compat() is never called and the strap stays a plain ANT+ 0x78
# sensor on the air.
#
# That distinction matters for P8b specifically. A strap being used to measure
# BLE coexistence must broadcast exactly what a commercial strap broadcasts;
# adding RadiANT beacon and attestation pages would change the airtime being
# measured, which is the one thing the phase is for. So the compat layer is
# LINKED and INERT, and the enrolment, private-mode, adaptive-frequency and
# command layers are not linked at all - they join when there is a reason.
target_sources(app PRIVATE
  ../radiant_core/src/profiles/profile_bits.c
  ../radiant_core/src/profiles/profile_common.c
  ../radiant_core/src/profiles/profile_sched.c
  ../radiant_core/src/profiles/profile_hr.c
  ../radiant_core/src/profiles/profile_schedule.c
  ../radiant_core/src/profiles/profile_compat.c
  ../radiant_core/src/profiles/profile_telemetry.c
  ../radiant_core/src/profiles/profile_handoff.c
)
target_include_directories(app PRIVATE ../radiant_core/src/profiles)

# ── radiant_core/src/node ───────────────────────────────────────────────────
#
# A LIBRARY, NOT AN APPLICATION - the plan's words, and the reason it can be
# linked here at all. The identity record decides the device number under
# ADR 0009, and the settings-backed NVM implementation is the one the dongle
# already uses.
target_sources(app PRIVATE
  ../radiant_core/src/node/node_ident.c
  ../radiant_core/src/node/node_nvm_settings.c
)
target_include_directories(app PRIVATE ../radiant_core/src/node)

# The antr_* declarations and the ANT wire constants. src/ is the application
# layer of the dongle, but these two headers are the interface radiant_core
# implements rather than anything dongle-specific.
target_include_directories(app PRIVATE ../src)

if(CONFIG_STRAP_BLE_HRS)
  target_sources(app PRIVATE src/strap_ble.c)
endif()
