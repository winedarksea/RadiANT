cmake_minimum_required(VERSION 3.20.0)

# RadiANT trainer: an FE-C bike trainer simulator with programmable ACK
# behaviour. See src/main.c for what it is, src/trainer_ackfault.h for the
# fault model, and docs/trainer-sim.md for how to drive it.
#
# Built like apps/treadmill - radiant and radiant/src/profiles, no sdk-ant -
# and like the treadmill it defaults to the nrf backend, never to null. See
# apps/treadmill/CMakeLists.txt and cmake/radiant_backend.cmake for why.
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/radiant_backend.cmake")
radiant_select_backend("${CMAKE_CURRENT_SOURCE_DIR}/../../radiant" "nrf" RADIANT_BACKEND_CONF)

set(trainer_backend_conf "${CMAKE_CURRENT_BINARY_DIR}/trainer_backend.conf")
file(WRITE "${trainer_backend_conf}"
"# Generated by apps/trainer/CMakeLists.txt from -DRADIANT_BACKEND. Do not edit.
CONFIG_RADIANT=y
CONFIG_RADIANT_ANTR_API=y
${RADIANT_BACKEND_CONF}
")
list(APPEND EXTRA_CONF_FILE "${trainer_backend_conf}")

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(radiant_trainer)

# ── No arbiter, by construction ─────────────────────────────────────────────
#
# This image exists to put an acknowledgement on the air at a CHOSEN instant.
# The MPSL gate wraps every transmit in an arbitration request and a
# follow-on reserve, which moves and sometimes refuses exactly that instant -
# so a trainer image with the gate in it would be measuring the gate. BLE is
# the usual way the gate arrives (radiant/Kconfig's nrf arm needs it under
# CONFIG_BT), so both are refused here rather than documented.
if(CONFIG_BT OR CONFIG_RADIANT_BACKEND_NRF_GATE_MPSL)
  message(FATAL_ERROR
    "trainer: CONFIG_BT / CONFIG_RADIANT_BACKEND_NRF_GATE_MPSL is set.\n"
    "  This simulator's whole job is to place an ACK at a chosen instant, and "
    "the MPSL arbiter moves that instant. Build it direct, with no BLE.")
endif()

# The instrument is the point of the image; an image without it is a plain
# trainer that cannot answer a single `ackfault` command.
if(NOT CONFIG_RADIANT_ACK_FAULTS)
  message(FATAL_ERROR
    "trainer: CONFIG_RADIANT_ACK_FAULTS is not set. prj.conf turns it on; "
    "something downstream turned it off.")
endif()

radiant_assert_backend("trainer")

target_sources(app PRIVATE
  src/main.c
  src/trainer_model.c
  src/trainer_ackfault.c
  src/trainer_shell.c
)

# The antr_* adapter, as every radiant application uses it.
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/sources.cmake")

# radiant/src/profiles: what the linker needs for the FE-C master, the same
# list apps/treadmill links. profile_fec.c (the DECODER) is absent there and
# absent here too - the trainer transmits page 25, it does not decode one.
target_sources(app PRIVATE
  ../../radiant/src/profiles/profile_bits.c
  ../../radiant/src/profiles/profile_common.c
  ../../radiant/src/profiles/profile_sched.c
  ../../radiant/src/profiles/profile_schedule.c
  ../../radiant/src/profiles/profile_telemetry.c
  ../../radiant/src/profiles/profile_handoff.c
  ../../radiant/src/profiles/profile_fec_tx.c
)
target_include_directories(app PRIVATE ../../radiant/src/profiles)

target_sources(app PRIVATE
  ../../radiant/src/node/node_ident.c
  ../../radiant/src/node/node_nvm_settings.c
)
target_include_directories(app PRIVATE ../../radiant/src/node)
