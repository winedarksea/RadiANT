cmake_minimum_required(VERSION 3.20.0)

# ANT+-only Zwift dongle - the headline product: minimum latency and loss,
# built for whichever of the boards in boards/ a user actually has. No BT, no
# Thread, no coexistence instrumentation - that is apps/dongle_thread's job,
# and stripping it out of this image entirely is deliberate, not an
# oversight: every byte and every interrupt priority here exists to answer
# ANT+ traffic as fast as possible, and a second radio stack even linked in
# and idle is one more thing between an interrupt firing and a slot closing.
#
# The sysbuild-cache reader and the RADIANT_BACKEND selection/assertion
# machinery are shared with every other RadiANT app (apps/hrm_ble,
# apps/dongle_thread) rather than copy-pasted - see that file for the
# sysbuild mechanics this depends on. The ANT_RADIO axis (sdk_ant vs core vs
# stub) is the ANT-serial-dongle-specific half of the same idea, and lives in
# apps/common/ instead, for the reason that file's own header gives.
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/radiant_backend.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/ant_radio_backend.cmake")

# radiant is appended unconditionally, and the asymmetry with sdk-ant is
# deliberate. sdk-ant is appended only for -DANT_RADIO=sdk_ant because it may
# not be on disk at all; radiant is in this repository and always is. So it
# is always a module and CONFIG_RADIANT gates it, which is both the ordinary
# Zephyr idiom and a better test: an sdk_ant build then asserts
# CONFIG_RADIANT=n positively, rather than proving nothing by the module's
# absence. radiant/CMakeLists.txt returns immediately when the symbol is n,
# so an sdk_ant build cannot reach a <radiant/...> header either.
#
# The HAL backend is resolved unconditionally too, even though it is only
# used when ANT_RADIO=core: resolving it once is simpler than threading it
# through every ANT_RADIO branch, and a stray RADIANT_BACKEND cache entry on
# an sdk_ant/stub build is harmless since nothing reads it.
radiant_select_backend("${CMAKE_CURRENT_SOURCE_DIR}/../../radiant" "null" RADIANT_BACKEND_CONF)
ant_select_radio("${CMAKE_CURRENT_SOURCE_DIR}" "" "${RADIANT_BACKEND_CONF}" ANT_RADIO_CONF)

# Write only on a change. Zephyr adds every merged fragment to
# CMAKE_CONFIGURE_DEPENDS, so rewriting an identical file each configure
# would hand ninja a freshly-stamped dependency of the configure step it just
# ran.
#
# It lands in the application binary directory because Zephyr globs *.conf
# there and merges the result *after* everything else, including
# -DEXTRA_CONF_FILE. That ordering is what makes this authoritative: the
# backend CMake actually compiled and the backend Kconfig records cannot
# disagree.
set(ANT_RADIO_CONF_FILE "${CMAKE_CURRENT_BINARY_DIR}/ant_radio_backend.conf")
set(ANT_RADIO_CONF_EXISTING "")
if(EXISTS "${ANT_RADIO_CONF_FILE}")
  file(READ "${ANT_RADIO_CONF_FILE}" ANT_RADIO_CONF_EXISTING)
endif()
if(NOT ANT_RADIO_CONF_EXISTING STREQUAL ANT_RADIO_CONF)
  file(WRITE "${ANT_RADIO_CONF_FILE}" "# Generated by CMakeLists.txt from -DANT_RADIO. Do not edit; do not commit.\n${ANT_RADIO_CONF}")
endif()

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(radiant_dongle)

# See apps/common/ant_radio_backend.cmake's ant_assert_radio() - only
# relevant when ANT_RADIO chose radiant in the first place, and it is the
# function's own job to check that, not this call site's.
ant_assert_radio("radiant_dongle")

target_sources(app PRIVATE src/main.c)

# ── apps/common: the antr_* seam, the serial bridge and its transports ─────
#
# One source list, added fresh into this application's own image - see that
# file's own header for why it is include()d rather than add_subdirectory()'d
# (a plain zephyr_library() there does not link). CONFIG_ANT_DONGLE_TRANSPORT_*
# (set by the choice in Kconfig) and
# CONFIG_ANT_DONGLE_RADIO_*/CONFIG_RADIANT_ANTR_API (set above by
# ant_select_radio() and by apps/common/Kconfig.antr_api's default) together
# decide what actually compiles from it into this image.
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/sources.cmake")

# ── The hostless node's identity (compat-C6) ────────────────────────────────
#
# docs/decisions/0009-hostless-node-identity.md. Default off and off in every
# dongle build: a dongle's epoch authority is the PC behind it, over 0xF3, and
# its pairing scalar arrives over 0xF5 - it wants none of this. The symbol and
# the wiring live here so that a node application built from this tree has a
# home for them, and so the storage backend is selected the same way every
# other backend in this project is.
#
# The counters' storage is a seam (radiant/src/node/node_nvm.h) with no
# default implementation, which is why the backend is a separate condition
# rather than being folded into the one above: a build that selects the
# identity and no backend must fail at the link, not fall back to RAM.
#
# radiant/src/node, not apps/common/node: this pair lives inside the module
# directory (radiant reaches no path outside itself, and this pair always
# did), but the decision to compile it belongs to this application, same as
# before the apps/ split - nothing in radiant/CMakeLists.txt reads either
# symbol.
if(CONFIG_RADIANT_NODE_IDENT)
  target_sources(app PRIVATE ../../radiant/src/node/node_ident.c)
  target_include_directories(app PRIVATE ../../radiant/src/node)
endif()
if(CONFIG_RADIANT_NODE_NVM_SETTINGS)
  target_sources(app PRIVATE ../../radiant/src/node/node_nvm_settings.c)
endif()
