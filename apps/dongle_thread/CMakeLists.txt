cmake_minimum_required(VERSION 3.20.0)

# The same ANT+ dongle as apps/dongle, plus Thread/Matter rebroadcast and the
# P3/P4 coexistence bench instruments (ble_coex_load.c, thread_coex_load.c)
# that apps/dongle deliberately carries none of - see that app's own
# CMakeLists.txt header for why splitting the two apart, rather than a
# Kconfig variant of one, is the point.
#
# The sysbuild-cache reader and the RADIANT_BACKEND selection/assertion
# machinery are shared with every other RadiANT app (apps/dongle,
# apps/hrm_ble) rather than copy-pasted - see cmake/radiant_backend.cmake for
# the sysbuild mechanics this depends on. The ANT_RADIO axis (sdk_ant vs core
# vs stub) is the ANT-serial-dongle-specific half of the same idea, and lives
# in apps/common/ instead, for the reason that file's own header gives.
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/radiant_backend.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/ant_radio_backend.cmake")

# radiant is appended unconditionally, same reasoning as apps/dongle's own
# CMakeLists.txt: it is in this repository and always is, so CONFIG_RADIANT
# gates it rather than the module's presence gating anything.
#
# THE P4 GATE NEEDS -DRADIANT_BACKEND=nrf ON THE COMMAND LINE, and the default
# here is still "null" - the same cheap-compile-check posture as apps/dongle,
# not "nrf" - because a script whose job is producing a bench image
# (scripts/build_p4.ps1) already passes -DRADIANT_BACKEND=nrf explicitly and
# reads the .config back to prove it landed; a default that silently disagreed
# with what a plain `west build` produced would be one more place for the
# arbitrated build to quietly become the null one.
radiant_select_backend("${CMAKE_CURRENT_SOURCE_DIR}/../../radiant" "null" RADIANT_BACKEND_CONF)
ant_select_radio("${CMAKE_CURRENT_SOURCE_DIR}" "" "${RADIANT_BACKEND_CONF}" ANT_RADIO_CONF)

# Write only on a change - see apps/dongle/CMakeLists.txt for the full
# rationale (CMAKE_CONFIGURE_DEPENDS churn on every configure otherwise).
set(ANT_RADIO_CONF_FILE "${CMAKE_CURRENT_BINARY_DIR}/ant_radio_backend.conf")
set(ANT_RADIO_CONF_EXISTING "")
if(EXISTS "${ANT_RADIO_CONF_FILE}")
  file(READ "${ANT_RADIO_CONF_FILE}" ANT_RADIO_CONF_EXISTING)
endif()
if(NOT ANT_RADIO_CONF_EXISTING STREQUAL ANT_RADIO_CONF)
  file(WRITE "${ANT_RADIO_CONF_FILE}" "# Generated by CMakeLists.txt from -DANT_RADIO. Do not edit; do not commit.\n${ANT_RADIO_CONF}")
endif()

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(radiant_dongle_thread)

# See apps/common/ant_radio_backend.cmake's ant_assert_radio() - only
# relevant when ANT_RADIO chose radiant in the first place.
ant_assert_radio("radiant_dongle_thread")

target_sources(app PRIVATE src/main.c)

# thread_coex_load.h and ble_coex_load.h sit at this application's own root
# (not under src/) - see the file-placement note in the reorg plan - so
# main.c's quoted #include needs this directory on its search path.
target_include_directories(app PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")

# ── The RadiANT bridge: the sample bus and everything on it ─────────────────
#
# docs/radiant-bridge.md sections 3-6 and 9. Five files from the module's
# src/bridge/ directory plus this application's own half under src/.
#
# The shape is copied from radiant/tests/CMakeLists.txt, which is the only
# working precedent in the tree for consuming this directory, and the
# zephyr_linker_sources() line below is the part of it that is NOT optional.
#
# RADIANT_SINK_DEFINE() (radiant_bridge.h) uses STRUCT_SECTION_ITERABLE, which
# emits each sink into a section named after the struct type. Emitting into a
# section and having that section PLACED - with the _radiant_sink_list_start /
# _end symbols STRUCT_SECTION_FOREACH() walks between - are two different
# things, and radiant_sink has no in-tree linker fragment of its own the way
# shell commands and bt_conn_cb do. Omit the line and the best case is an
# undefined reference at the final link; the worse case is that it links and
# STRUCT_SECTION_FOREACH() silently iterates ZERO sinks, which is not a
# compile error, not a link error, and looks from the console exactly like
# "no sensors in range". src/bridge_pump.c logs the count at boot for that
# reason: "bridge: 0 sinks" is the failure announcing itself.
#
# radiant_matter.c and radiant_rd_adapter.c are deliberately NOT here. Both
# compile and are tested (radiant/tests builds them), but Matter needs
# connectedhomeip - which is a flash-budget spike this application has not run
# - and the running-dynamics adapter has no bench evidence behind it yet.
# Linking either would make an untested path look like a shipped one.
if(CONFIG_RADIANT_BRIDGE)
  target_sources(app PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/../../radiant/src/bridge/radiant_bridge.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../radiant/src/bridge/radiant_binding.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../radiant/src/bridge/radiant_hr_adapter.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../radiant/src/bridge/radiant_rules.c"
    # Liveness lands with the bus, not after a sink. RADIANT_SAMPLE_STALE has
    # been defined and honoured since P6 with nothing producing it, and a sink
    # written against a bus that never marks anything stale grows that
    # assumption into itself.
    "${CMAKE_CURRENT_SOURCE_DIR}/../../radiant/src/bridge/radiant_liveness.c"
    # The application half: the drain pump and the 1 Hz liveness clock.
    "${CMAKE_CURRENT_SOURCE_DIR}/src/bridge_pump.c"
  )
  target_include_directories(app PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/../../radiant/src/bridge"
    "${CMAKE_CURRENT_SOURCE_DIR}/src")
  zephyr_linker_sources(SECTIONS
    "${CMAKE_CURRENT_SOURCE_DIR}/../../radiant/src/bridge/radiant_bridge_sections.ld")
endif()

target_sources_ifdef(CONFIG_ANT_DONGLE_RX_TAP app PRIVATE
  "${CMAKE_CURRENT_SOURCE_DIR}/src/rx_tap.c")
