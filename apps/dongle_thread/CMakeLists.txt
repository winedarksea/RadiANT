cmake_minimum_required(VERSION 3.20.0)

# The same ANT+ dongle as apps/dongle, plus Thread/Matter rebroadcast and the
# P3/P4 coexistence bench instruments (ble_coex_load.c, thread_coex_load.c)
# that apps/dongle deliberately carries none of - see that app's own
# CMakeLists.txt header for why splitting the two apart, rather than a
# Kconfig variant of one, is the point.
#
# The sysbuild-cache reader and the RADIANT_BACKEND selection/assertion
# machinery are shared with every other RadiANT app (apps/dongle,
# apps/hrm_ble) rather than copy-pasted - see cmake/radiant_backend.cmake for
# the sysbuild mechanics this depends on. The ANT_RADIO axis (sdk_ant vs core
# vs stub) is the ANT-serial-dongle-specific half of the same idea, and lives
# in apps/common/ instead, for the reason that file's own header gives.
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/radiant_backend.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/ant_radio_backend.cmake")

# radiant is appended unconditionally, same reasoning as apps/dongle's own
# CMakeLists.txt: it is in this repository and always is, so CONFIG_RADIANT
# gates it rather than the module's presence gating anything.
#
# THE P4 GATE NEEDS -DRADIANT_BACKEND=nrf ON THE COMMAND LINE, and the default
# here is still "null" - the same cheap-compile-check posture as apps/dongle,
# not "nrf" - because a script whose job is producing a bench image
# (scripts/build_p4.ps1) already passes -DRADIANT_BACKEND=nrf explicitly and
# reads the .config back to prove it landed; a default that silently disagreed
# with what a plain `west build` produced would be one more place for the
# arbitrated build to quietly become the null one.
radiant_select_backend("${CMAKE_CURRENT_SOURCE_DIR}/../../radiant" "null" RADIANT_BACKEND_CONF)
ant_select_radio("${CMAKE_CURRENT_SOURCE_DIR}" "" "${RADIANT_BACKEND_CONF}" ANT_RADIO_CONF)

# Write only on a change - see apps/dongle/CMakeLists.txt for the full
# rationale (CMAKE_CONFIGURE_DEPENDS churn on every configure otherwise).
set(ANT_RADIO_CONF_FILE "${CMAKE_CURRENT_BINARY_DIR}/ant_radio_backend.conf")
set(ANT_RADIO_CONF_EXISTING "")
if(EXISTS "${ANT_RADIO_CONF_FILE}")
  file(READ "${ANT_RADIO_CONF_FILE}" ANT_RADIO_CONF_EXISTING)
endif()
if(NOT ANT_RADIO_CONF_EXISTING STREQUAL ANT_RADIO_CONF)
  file(WRITE "${ANT_RADIO_CONF_FILE}" "# Generated by CMakeLists.txt from -DANT_RADIO. Do not edit; do not commit.\n${ANT_RADIO_CONF}")
endif()

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(radiant_dongle_thread)

# See apps/common/ant_radio_backend.cmake's ant_assert_radio() - only
# relevant when ANT_RADIO chose radiant in the first place.
ant_assert_radio("radiant_dongle_thread")

target_sources(app PRIVATE src/main.c)

# thread_coex_load.h and ble_coex_load.h sit at this application's own root
# (not under src/) - see the file-placement note in the reorg plan - so
# main.c's quoted #include needs this directory on its search path.
target_include_directories(app PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")

# ── apps/common: the antr_* seam, the serial bridge and its transports ─────
#
# One source list, added fresh into this application's own image - see that
# file's own header for why it is include()d rather than add_subdirectory()'d.
# CONFIG_ANT_DONGLE_TRANSPORT_* (set by the choice in Kconfig) and
# CONFIG_ANT_DONGLE_RADIO_*/CONFIG_RADIANT_ANTR_API (set above by
# ant_select_radio() and by apps/common/Kconfig.antr_api's default) together
# decide what actually compiles from it into this image.
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/sources.cmake")

# ── The P3/P4 coexistence bench instruments ─────────────────────────────────
#
# The one thing that makes this application apps/dongle_thread rather than a
# second copy of apps/dongle: a second wireless stack that actually wants the
# radio. Off by default in every shipping image (see each Kconfig's own
# help), so these two lines are inert everywhere except a bench build that
# explicitly turns one on.
target_sources_ifdef(CONFIG_ANT_DONGLE_BLE_COEX_LOAD app PRIVATE
  "${CMAKE_CURRENT_SOURCE_DIR}/ble_coex_load.c")
target_sources_ifdef(CONFIG_ANT_DONGLE_THREAD_COEX_LOAD app PRIVATE
  "${CMAKE_CURRENT_SOURCE_DIR}/thread_coex_load.c")

# ── The hostless node's identity (compat-C6) ────────────────────────────────
#
# Same wiring as apps/dongle's own CMakeLists.txt - see that file's comment
# for the full rationale. Off by default here too: this application's epoch
# authority is still the PC behind it, same as the plain dongle.
if(CONFIG_RADIANT_NODE_IDENT)
  target_sources(app PRIVATE ../../radiant/src/node/node_ident.c)
  target_include_directories(app PRIVATE ../../radiant/src/node)
endif()
if(CONFIG_RADIANT_NODE_NVM_SETTINGS)
  target_sources(app PRIVATE ../../radiant/src/node/node_nvm_settings.c)
endif()
