cmake_minimum_required(VERSION 3.20.0)

# RadiANT smart trainer: the mixed ANT+ FE-C / BLE FTMS reference design for a
# controllable bike trainer. See src/main.c for what it is, src/trainer_source.h
# for the one seam a manufacturer implements, and
# docs/smart-trainer-reference-design.md for the porting guide.
#
# Built like apps/treadmill: radiant and radiant/src/profiles, no sdk-ant, the
# nrf backend by default (never null - see cmake/radiant_backend.cmake).
#
# NOT apps/trainer. That image is an ACK-fault instrument and refuses BLE by
# construction; this one is a product and carries BLE through the MPSL gate.
# It shares exactly one file with it: apps/trainer/src/trainer_model.c, the
# FE-C SS8.8.4 physics behind the simulated source, compiled from there below
# rather than copied.
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/radiant_backend.cmake")
radiant_select_backend("${CMAKE_CURRENT_SOURCE_DIR}/../../radiant" "nrf" RADIANT_BACKEND_CONF)

set(smart_trainer_backend_conf "${CMAKE_CURRENT_BINARY_DIR}/smart_trainer_backend.conf")
file(WRITE "${smart_trainer_backend_conf}"
"# Generated by apps/smart_trainer/CMakeLists.txt from -DRADIANT_BACKEND. Do not edit.
CONFIG_RADIANT=y
CONFIG_RADIANT_ANT_API=y
${RADIANT_BACKEND_CONF}
")
list(APPEND EXTRA_CONF_FILE "${smart_trainer_backend_conf}")

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(radiant_smart_trainer)

# The BLE interlock from the other side, as apps/treadmill's: CONFIG_BT without
# SMART_TRAINER_BLE (which selects the MPSL gate) makes radiant's nrf arm
# unselectable, and Kconfig silently falls back to the null backend.
if(CONFIG_BT AND NOT CONFIG_SMART_TRAINER_BLE)
  message(FATAL_ERROR
    "smart_trainer: CONFIG_BT is set but CONFIG_SMART_TRAINER_BLE is not.\n"
    "  Without the MPSL gate the nrf backend is unselectable and Kconfig "
    "falls back to the null backend - a node that puts nothing on the air.\n"
    "  Turn BLE on through this application's own symbol:\n"
    "    west build ... -- -Dsmart_trainer_EXTRA_CONF_FILE=ble.conf\n"
    "  THE PREFIX IS THE BASENAME OF THIS DIRECTORY (sysbuild names the image "
    "after it); a wrong prefix is accepted in silence.")
endif()

radiant_assert_backend("smart_trainer")

target_include_directories(app PRIVATE src ../trainer/src)

target_sources(app PRIVATE
  src/main.c
  src/trainer_source.c
  # Kernel-free: tests/glue compiles these three.
  src/trainer_equip.c
  src/trainer_cmd.c
  src/trainer_sim_model.c
  # apps/trainer's FE-C SS8.8.4 model, reused in place (trainer_sim_model.c
  # wraps it).
  ../trainer/src/trainer_model.c
)

if(CONFIG_SMART_TRAINER_SOURCE_SIMULATED)
  target_sources(app PRIVATE src/trainer_sim.c)
else()
  target_sources(app PRIVATE src/trainer_source_custom.c)
endif()

if(CONFIG_SMART_TRAINER_BLE)
  target_sources(app PRIVATE src/trainer_ble.c)
endif()

include("${CMAKE_CURRENT_SOURCE_DIR}/../common/sources.cmake")

# radiant/src/profiles: the FE-C master and what the slot engine needs - the
# same list apps/treadmill and apps/trainer link.
target_sources(app PRIVATE
  ../../radiant/src/profiles/radiant_profile_bits.c
  ../../radiant/src/profiles/radiant_profile_common.c
  ../../radiant/src/profiles/radiant_profile_sched.c
  ../../radiant/src/profiles/radiant_profile_schedule.c
  ../../radiant/src/profiles/radiant_profile_telemetry.c
  ../../radiant/src/profiles/radiant_profile_handoff.c
  ../../radiant/src/profiles/radiant_profile_fec_tx.c
)

target_sources(app PRIVATE
  ../../radiant/src/node/radiant_node_ident.c
  ../../radiant/src/node/radiant_node_nvm_settings.c
)
