cmake_minimum_required(VERSION 3.20.0)

# RadiANT treadmill: a node, not a dongle, and the second manufacturer
# reference design in this tree. See src/main.c for what it is,
# src/treadmill_source.h for the one seam a manufacturer implements, Kconfig for
# what it is allowed to do, and docs/treadmill-reference-design.md for the
# porting guide.
#
# LIKE apps/hrm_ble AND UNLIKE apps/sim, THIS NEEDS NO sdk-ant AT ALL. It is
# built out of radiant and radiant/src/profiles/, both of which are in this
# repository.
#
# THE DEFAULT IS nrf AND NOT null, and the asymmetry with radiant's own CMake
# default is the same one scripts/build_all.ps1 documents: null is right for a
# compile check and wrong for a script whose job is producing images somebody
# flashes. A null-radio node boots, logs "transmitting", and puts nothing on the
# air - indistinguishable from a dead antenna, and it has cost this project a
# Feather flash and a whole session once already.
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/radiant_backend.cmake")
radiant_select_backend("${CMAKE_CURRENT_SOURCE_DIR}/../../radiant" "nrf" RADIANT_BACKEND_CONF)

set(treadmill_backend_conf "${CMAKE_CURRENT_BINARY_DIR}/treadmill_backend.conf")
file(WRITE "${treadmill_backend_conf}"
"# Generated by apps/treadmill/CMakeLists.txt from -DRADIANT_BACKEND. Do not edit.
CONFIG_RADIANT=y
CONFIG_RADIANT_ANTR_API=y
${RADIANT_BACKEND_CONF}
")
list(APPEND EXTRA_CONF_FILE "${treadmill_backend_conf}")

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(radiant_treadmill)

# ── The BLE interlock, from the other side ──────────────────────────────────
#
# TREADMILL_BLE `select`s RADIANT_BACKEND_NRF_GATE_MPSL, which covers the route
# through this application's own Kconfig. This covers the other route: any conf
# fragment, board file or -DCONFIG_ that turns CONFIG_BT on WITHOUT going
# through TREADMILL_BLE.
#
# That combination is not merely unsupported, it is silent. radiant/Kconfig's
# nrf arm carries `depends on !BT || ..._GATE_MPSL`, so CONFIG_BT=y with no gate
# makes the arm unselectable and Kconfig falls back to the choice default - the
# NULL backend. The resulting node boots, logs "transmitting", and puts nothing
# on the air.
#
# BEFORE radiant_assert_backend(), NOT AFTER, and the ordering was measured
# rather than assumed - see apps/hrm_ble/CMakeLists.txt, which records the
# experiment. The generic assert's message is about a CMake cache variable and
# lists three unrelated `depends on` symbols; the specific message has to come
# first to be the one anybody reads.
if(CONFIG_BT AND NOT CONFIG_TREADMILL_BLE)
  message(FATAL_ERROR
    "treadmill: CONFIG_BT is set but CONFIG_TREADMILL_BLE is not.\n"
    "  radiant/Kconfig's nrf backend has `depends on !BT || "
    "RADIANT_BACKEND_NRF_GATE_MPSL`, so BT without the gate makes the nrf arm "
    "unselectable and Kconfig silently falls back to the null backend - a node "
    "that boots, logs 'transmitting' and puts nothing on the air.\n"
    "  Turn BLE on through this application's own symbol, which selects the "
    "gate for you:\n"
    "    west build ... -- -Dtreadmill_EXTRA_CONF_FILE=ble.conf\n"
    "  THE PREFIX IS THE BASENAME OF THIS DIRECTORY. Sysbuild names an image "
    "after the basename of APP_DIR, so a wrong prefix names an image that does "
    "not exist and is accepted in silence - apps/hrm_ble shipped exactly that "
    "no-op for a while after a directory rename, and two builds with and "
    "without the flag produced byte-identical .config files.")
endif()

# The backend actually compiled is the backend that was asked for. See
# cmake/radiant_backend.cmake for the full rationale.
radiant_assert_backend("treadmill")

target_sources(app PRIVATE
  src/main.c
  src/treadmill_source.c
  src/treadmill_state.c
)

# ── The treadmill source ────────────────────────────────────────────────────
#
# A Kconfig `choice`, so exactly one of these is ever compiled and exactly one
# treadmill_source_register() call is ever linked. CONFIG_TREADMILL_SOURCE_CUSTOM
# compiles the stub, which registers nothing on purpose: see that symbol's help
# for why there is deliberately no fallback to the simulator.
if(CONFIG_TREADMILL_SOURCE_SIMULATED)
  target_sources(app PRIVATE src/treadmill_sim.c)
elseif(CONFIG_TREADMILL_SOURCE_CUSTOM)
  target_sources(app PRIVATE src/treadmill_source_custom.c)
endif()

if(CONFIG_TREADMILL_BLE)
  target_sources(app PRIVATE src/treadmill_ble.c)
endif()

if(CONFIG_TREADMILL_HR_RX)
  target_sources(app PRIVATE src/treadmill_hr_rx.c)
endif()

# ── The antr_* adapter ──────────────────────────────────────────────────────
#
# apps/common/ant_radio_radiant.c is what implements antr_init() and the rest of
# apps/common/ant/ant_radio.h on top of radiant's modules. The generated backend
# fragment above forces CONFIG_RADIANT_ANTR_API=y unconditionally.
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/sources.cmake")

# ── radiant/src/profiles ────────────────────────────────────────────────────
#
# WHAT THE LINKER SAYS IS NEEDED, WHICH IS MORE THAN THE PAGES NEED.
#
# profile_fec_tx.c and profile_sdm.c are the two masters. profile_common.c is
# pages 80 and 81. profile_sched.c is the slot engine both masters drive, and it
# pulls in profile_schedule.c and profile_telemetry.c for the descriptor path -
# which NEITHER master uses (both configure cfg.desc = NULL, because an ANT+
# compatibility type announces no schema) but which the engine references
# unconditionally. profile_bits.c is under those.
#
# profile_compat.c is deliberately ABSENT, unlike in apps/hrm_ble. That
# application links the compat client inert so that a coexistence measurement is
# taken against a strap broadcasting exactly what a commercial strap
# broadcasts. Here there is no equivalent argument and no compat allocation for
# device types 0x11 or 0x7C - docs/decisions/0008's two page numbers are
# registered on 0x0B and 0x78 only - so linking it would be linking a layer
# with no page numbers to put anything on.
#
# profile_fec.c (the DECODER) is also absent, and that absence is the point of
# the two-translation-unit split: a treadmill transmits and does not decode
# another machine's pages.
target_sources(app PRIVATE
  ../../radiant/src/profiles/profile_bits.c
  ../../radiant/src/profiles/profile_common.c
  ../../radiant/src/profiles/profile_sched.c
  ../../radiant/src/profiles/profile_schedule.c
  ../../radiant/src/profiles/profile_telemetry.c
  ../../radiant/src/profiles/profile_handoff.c
  ../../radiant/src/profiles/profile_fec_tx.c
)
if(CONFIG_TREADMILL_ANT_SDM)
  target_sources(app PRIVATE ../../radiant/src/profiles/profile_sdm.c)
endif()
if(CONFIG_TREADMILL_HR_RX)
  # Only the header is needed for the one byte the slave reads (see
  # src/treadmill_hr_rx.c), but profile_hr.c is what defines
  # PROFILE_HR_INVALID_BPM's neighbours and the encoders a future console
  # would want. Compiled so the constants and the profile stay one file.
  target_sources(app PRIVATE ../../radiant/src/profiles/profile_hr.c
                             ../../radiant/src/profiles/profile_compat.c)
endif()
target_include_directories(app PRIVATE ../../radiant/src/profiles)

# ── radiant/src/node ────────────────────────────────────────────────────────
#
# The identity record decides the device number under ADR 0009, and the
# settings-backed NVM implementation is the one the dongle already uses.
target_sources(app PRIVATE
  ../../radiant/src/node/node_ident.c
  ../../radiant/src/node/node_nvm_settings.c
)
target_include_directories(app PRIVATE ../../radiant/src/node)
