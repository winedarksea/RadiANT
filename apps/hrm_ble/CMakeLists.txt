cmake_minimum_required(VERSION 3.20.0)

# RadiANT heart-rate node: a node, not a dongle, and the manufacturer reference
# design. See src/main.c for why it exists, src/hrm_sensor.h for the one seam a
# manufacturer implements, Kconfig for what it is allowed to do, and
# docs/hrm-reference-design.md for the porting guide.
#
# UNLIKE sim/, THIS NEEDS NO sdk-ant AT ALL. sim/ is built entirely out of
# sdk-ant's profile libraries and its radio, and says so in a FATAL_ERROR;
# this one is built out of radiant and radiant/src/profiles/, both
# of which are in this repository. That is the whole point of it: it is the
# first image where
# the clean-room stack transmits as a master with a real ANT+ profile on top,
# and a dependency on the proven stack would defeat the demonstration.
#
# radiant is appended unconditionally and its backend chosen with
# -DRADIANT_BACKEND, exactly as apps/dongle's CMakeLists.txt does - both now
# through the shared cmake/radiant_backend.cmake, included below. The
# duplication of *calling* it (rather than sharing one CMakeLists.txt) is
# deliberate and matches apps/sim's note: `west build apps/hrm_ble` must not
# depend on the dongle's build files existing or being current.
#
# THE DEFAULT IS nrf AND NOT null, and the asymmetry with radiant's own
# CMake default is the same one scripts/build_all.ps1 documents: null is right
# for a compile check and wrong for a script whose job is producing images
# somebody flashes. A null-radio strap boots, logs "transmitting", and puts
# nothing on the air - indistinguishable from a dead antenna, and it has cost
# this project a Feather flash and a whole session once already.
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/radiant_backend.cmake")
radiant_select_backend("${CMAKE_CURRENT_SOURCE_DIR}/../../radiant" "nrf" RADIANT_BACKEND_CONF)

set(hrm_backend_conf "${CMAKE_CURRENT_BINARY_DIR}/hrm_backend.conf")
file(WRITE "${hrm_backend_conf}"
"# Generated by apps/hrm_ble/CMakeLists.txt from -DRADIANT_BACKEND. Do not edit.
CONFIG_RADIANT=y
CONFIG_RADIANT_ANTR_API=y
${RADIANT_BACKEND_CONF}
")
list(APPEND EXTRA_CONF_FILE "${hrm_backend_conf}")

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(radiant_hrm_ble)

# ── The BLE interlock, from the other side ──────────────────────────────────
#
# HRM_BLE_HRS `select`s RADIANT_BACKEND_NRF_GATE_MPSL, which covers the route
# through this application's own Kconfig. This covers the other route: any conf
# fragment, board file or -DCONFIG_ that turns CONFIG_BT on WITHOUT going
# through HRM_BLE_HRS.
#
# That combination is not merely unsupported, it is silent.
# radiant/Kconfig's nrf arm carries `depends on !BT || ..._GATE_MPSL`, so
# CONFIG_BT=y with no gate makes the arm unselectable and Kconfig falls back to
# the choice default - the NULL backend. The resulting node boots, logs
# "transmitting", and puts nothing on the air.
#
# BEFORE radiant_assert_backend(), NOT AFTER, and the ordering was measured
# rather than assumed. A build of this application with CONFIG_BT=y and no gate
# does already fail - but at the generic assert, whose message is about a CMake
# cache variable and lists three unrelated `depends on` symbols to go and check.
# The specific message has to come first to be the one anybody reads. (Verified
# by building exactly that configuration: with the two blocks the other way
# round, the output was radiant_assert_backend()'s "Kconfig chose
# CONFIG_RADIANT_BACKEND_NULL instead" and this text never appeared.)
#
# It also catches a case the assert structurally cannot see: a build that asked
# for `null` on purpose and then enabled BT.
if(CONFIG_BT AND NOT CONFIG_HRM_BLE_HRS)
  message(FATAL_ERROR
    "hrm_ble: CONFIG_BT is set but CONFIG_HRM_BLE_HRS is not.\n"
    "  radiant/Kconfig's nrf backend has `depends on !BT || "
    "RADIANT_BACKEND_NRF_GATE_MPSL`, so BT without the gate makes the nrf arm "
    "unselectable and Kconfig silently falls back to the null backend - a node "
    "that boots, logs 'transmitting' and puts nothing on the air.\n"
    "  Turn BLE on through this application's own symbol, which selects the "
    "gate for you:\n"
    "    west build ... -- -Dhrm_ble_EXTRA_CONF_FILE=ble.conf\n"
    "  THE PREFIX IS THE BASENAME OF THIS DIRECTORY. Sysbuild names an image "
    "after the basename of APP_DIR, so a stale `-Dstrap_EXTRA_CONF_FILE=` "
    "names an image that does not exist and is accepted in silence.")
endif()

# The backend actually compiled is the backend that was asked for.
#
# radiant_assert_backend(), shared with apps/dongle's CMakeLists.txt - see
# cmake/radiant_backend.cmake for the full rationale. What it guards against
# is a Kconfig `depends on` going silently unmet, falling back to the null
# backend: a null-radio node boots, logs "transmitting", and puts nothing on
# the air. It has happened twice on the Nordic side, and is more likely on TI,
# where the cc26xx arm has to depend on SOC_SERIES_CC13X2_CC26X2 because no
# SOC_COMPATIBLE_* symbol exists for the part.
radiant_assert_backend("hrm_ble")

target_sources(app PRIVATE
  src/main.c
  src/hrm_sensor.c
)

# ── The heart-rate source ───────────────────────────────────────────────────
#
# A Kconfig `choice`, so exactly one of these is ever compiled and exactly one
# hrm_sensor_register() call is ever linked. CONFIG_HRM_SENSOR_CUSTOM compiles
# none of them on purpose: see that symbol's help for why there is deliberately
# no fallback to the simulator.
#
# hrm_sim_sched.c is listed under the simulator with hrm_sim.c, but it is also
# compiled on its own by apps/hrm_ble/tests/sim_sched - it is a pure function
# with no kernel dependencies precisely so that it can be.
if(CONFIG_HRM_SENSOR_SIMULATED)
  target_sources(app PRIVATE src/hrm_sim.c src/hrm_sim_sched.c)
elseif(CONFIG_HRM_SENSOR_EXAMPLE)
  target_sources(app PRIVATE src/hrm_sensor_example.c)
endif()

# ── The antr_* adapter ──────────────────────────────────────────────────────
#
# apps/common/ant_radio_radiant.c (radiant_api.c, before it moved out of
# radiant) is what actually implements antr_init() and the rest of
# apps/common/ant/ant_radio.h on top of radiant's modules -
# CONFIG_RADIANT=y alone no longer pulls it in, since it is application-side
# now (see apps/common/Kconfig.antr_api). hrm_backend_conf above forces
# CONFIG_RADIANT_ANTR_API=y unconditionally, so this is never conditional in
# practice, but apps/common/sources.cmake's _ifdef-gated sources are shared
# with apps/dongle verbatim anyway, so nothing here fails silently if that
# ever changes.
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/sources.cmake")

# ── radiant/src/profiles ───────────────────────────────────────────────
#
# THE FIRST NON-TEST CONSUMER OF THESE FILES. Until this application they were
# compiled only by radiant/tests/CMakeLists.txt, so every profile in the
# project had only ever run inside a unit test. Nothing in them needed changing
# to run for real, which is the claim this build makes and the reason it is
# worth making.
#
# WHAT THE LINKER SAYS IS NEEDED, WHICH IS MORE THAN THE PAGE NEEDS.
#
# The first four are the heart-rate page itself. The other two are pulled in by
# the seam rather than by the profile: profile_sched.c's slot rotation calls
# into profile_schedule.c to encode the descriptor, and profile_hr_next() calls
# profile_compat_next() unconditionally - the compat client is a *seam*, and
# whether it emits anything is a run-time question of whether a compat context
# was attached, not a compile-time one. This application attaches none, so
# profile_hr_set_compat() is never called and the strap stays a plain ANT+ 0x78
# sensor on the air.
#
# That distinction matters for P8b specifically. A strap being used to measure
# BLE coexistence must broadcast exactly what a commercial strap broadcasts;
# adding RadiANT beacon and attestation pages would change the airtime being
# measured, which is the one thing the phase is for. So the compat layer is
# LINKED and INERT, and the enrolment, private-mode, adaptive-frequency and
# command layers are not linked at all - they join when there is a reason.
target_sources(app PRIVATE
  ../../radiant/src/profiles/profile_bits.c
  ../../radiant/src/profiles/profile_common.c
  ../../radiant/src/profiles/profile_sched.c
  ../../radiant/src/profiles/profile_hr.c
  ../../radiant/src/profiles/profile_schedule.c
  ../../radiant/src/profiles/profile_compat.c
  ../../radiant/src/profiles/profile_telemetry.c
  ../../radiant/src/profiles/profile_handoff.c
)
target_include_directories(app PRIVATE ../../radiant/src/profiles)

# ── radiant/src/node ───────────────────────────────────────────────────
#
# A LIBRARY, NOT AN APPLICATION - the plan's words, and the reason it can be
# linked here at all. The identity record decides the device number under
# ADR 0009, and the settings-backed NVM implementation is the one the dongle
# already uses.
target_sources(app PRIVATE
  ../../radiant/src/node/node_ident.c
  ../../radiant/src/node/node_nvm_settings.c
)
target_include_directories(app PRIVATE ../../radiant/src/node)

if(CONFIG_HRM_BLE_HRS)
  target_sources(app PRIVATE src/hrm_ble.c)
endif()
