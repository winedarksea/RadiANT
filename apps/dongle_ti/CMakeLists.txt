cmake_minimum_required(VERSION 3.20.0)

# ANT+ dongle on the second vendor's silicon - a CC13x2/CC26x2 LaunchXL. See
# boards/cc26x2r1_launchxl.conf for what this board can and cannot do (no USB
# device peripheral at all, so the ANT serial protocol goes out a UART).
#
# UNVERIFIED PENDING HARDWARE beyond a single bring-up bench: the coexistence
# arm (ti_coex.conf, CONFIG_RADIANT_COEX154_CC26XX_FORK) builds but has never
# run against a real 802.15.4 peer. See radiant/coex154_ti/ and
# ti_coex.conf's own header before recording anything off that arm. This is
# recorded in the README, not just here.
#
# The sysbuild-cache reader and the RADIANT_BACKEND selection/assertion
# machinery are shared with every other RadiANT app (apps/dongle,
# apps/dongle_thread, apps/hrm_ble) - see cmake/radiant_backend.cmake for the
# sysbuild mechanics this depends on. The ANT_RADIO axis lives in
# apps/common/ instead, same reason as every other ANT-serial application.
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/radiant_backend.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/ant_radio_backend.cmake")

# THE DEFAULT IS "cc26xx" - the only backend that means anything on this
# silicon. RADIANT_BACKEND_NRF names a TIMER this part does not have and could
# never configure here, so cc26xx is not merely the useful default, it is the
# only real one.
#
# It used to be "null", to keep a bare `west build apps/dongle_ti` configuring
# and linking with no -DEXTRA_ZEPHYR_MODULES and no hal_ti checkout. That
# property is exactly what is being given up, and deliberately: what it bought
# was a build that succeeds while producing an image whose radio refuses every
# arm. A bare build that now fails for want of hal_ti says so immediately;
# a bare build that used to succeed said nothing, and the resulting image is
# indistinguishable from a dead antenna until somebody spends a bench session
# on it.
#
# Every real build of this application already passes -DRADIANT_BACKEND=cc26xx
# explicitly (see scripts/build_all.ps1's TI rows and scripts/flash_ti.ps1),
# so this changes no build anybody runs - and the .config read-back after the
# build is still what actually proves which backend was compiled, see
# radiant_assert_backend() below.
radiant_select_backend("${CMAKE_CURRENT_SOURCE_DIR}/../../radiant" "cc26xx" RADIANT_BACKEND_CONF)
ant_select_radio("${CMAKE_CURRENT_SOURCE_DIR}" "" "${RADIANT_BACKEND_CONF}" ANT_RADIO_CONF)

# Write only on a change - see apps/dongle/CMakeLists.txt for the full
# rationale.
set(ANT_RADIO_CONF_FILE "${CMAKE_CURRENT_BINARY_DIR}/ant_radio_backend.conf")
set(ANT_RADIO_CONF_EXISTING "")
if(EXISTS "${ANT_RADIO_CONF_FILE}")
  file(READ "${ANT_RADIO_CONF_FILE}" ANT_RADIO_CONF_EXISTING)
endif()
if(NOT ANT_RADIO_CONF_EXISTING STREQUAL ANT_RADIO_CONF)
  file(WRITE "${ANT_RADIO_CONF_FILE}" "# Generated by CMakeLists.txt from -DANT_RADIO. Do not edit; do not commit.\n${ANT_RADIO_CONF}")
endif()

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(radiant_dongle_ti)

ant_assert_radio("radiant_dongle_ti")

target_sources(app PRIVATE src/main.c)

# ── apps/common: the antr_* seam, the serial bridge and its transports ─────
#
# Same file, same reasoning as apps/dongle and apps/dongle_thread. This board
# always lands on CONFIG_ANT_DONGLE_TRANSPORT_UART (see boards/*.conf - there
# is no USB device peripheral on this part at all), so only
# apps/common/ant_uart_transport.c actually compiles out of the transport
# trio.
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/sources.cmake")

# ── The hostless node's identity (compat-C6) ────────────────────────────────
#
# Same wiring as apps/dongle's and apps/dongle_thread's own CMakeLists.txt.
# Off by default here too.
if(CONFIG_RADIANT_NODE_IDENT)
  target_sources(app PRIVATE ../../radiant/src/node/node_ident.c)
  target_include_directories(app PRIVATE ../../radiant/src/node)
endif()
if(CONFIG_RADIANT_NODE_NVM_SETTINGS)
  target_sources(app PRIVATE ../../radiant/src/node/node_nvm_settings.c)
endif()
